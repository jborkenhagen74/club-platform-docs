cmake_minimum_required(VERSION 3.25)
project(ClubPlatformHelloExtension LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(hello-extension SHARED hello_extension.cpp)
target_include_directories(hello-extension PRIVATE ../../../sdk)
if(MSVC)
    target_compile_options(hello-extension PRIVATE /W4 /permissive-)
else()
    target_compile_options(hello-extension PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(check-extension check_extension.cpp)
    target_include_directories(check-extension PRIVATE ../../../sdk)
    target_link_libraries(check-extension PRIVATE ${CMAKE_DL_LIBS})
    add_test(NAME abi2-load COMMAND check-extension $<TARGET_FILE:hello-extension>)
endif()
